// ------------------------------------------------------- merge analysis ----

export function ancestorSet(db, repoId, headCommitId) {
  const set = new Set();
  let current = headCommitId;
  while (current && !set.has(current)) {
    set.add(current);
    const c = commitRow(db, current);
    if (!c || c.repo_id !== repoId) break;
    current = c.parent_id;
  }
  return set;
}

export function mergeBaseCommit(db, repoId, a, b) {
  if (!a || !b) return null;
  const ancestorsOfA = new Set();
  let current = a;
  while (current && !ancestorsOfA.has(current)) {
    ancestorsOfA.add(current);
    const c = commitRow(db, current);
    if (!c || c.repo_id !== repoId) break;
    current = c.parent_id;
  }
  current = b;
  const seen = new Set();
  while (current && !seen.has(current)) {
    seen.add(current);
    if (ancestorsOfA.has(current)) return current;
    const c = commitRow(db, current);
    if (!c || c.repo_id !== repoId) break;
    current = c.parent_id;
  }
  return null;
}

// Three-way analysis of merging `compareCommitId` into `baseHeadId`.
export function analyzeMerge(db, repoId, baseHeadId, compareCommitId) {
  const baseTree = getTree(db, baseHeadId);
  const compareTree = getTree(db, compareCommitId);
  const base = mergeBaseCommit(db, repoId, baseHeadId, compareCommitId);
  const baseCommonTree = getTree(db, base);   // empty when no common ancestor
  const conflicts = [];
  const merged = new Map(baseTree);
  const paths = new Set([...baseCommonTree.keys(), ...baseTree.keys(), ...compareTree.keys()]);
  for (const path of [...paths].sort()) {
    const common = baseCommonTree.has(path) ? baseCommonTree.get(path) : null;
    const ours = baseTree.has(path) ? baseTree.get(path) : null;
    const theirs = compareTree.has(path) ? compareTree.get(path) : null;
    const oursChanged = ours !== common;
    const theirsChanged = theirs !== common;
    if (!theirsChanged) continue;
    if (oursChanged && ours !== theirs) { conflicts.push(path); continue; }
    if (theirs === null) merged.delete(path); else merged.set(path, theirs);
  }
  return { clean: conflicts.length === 0, conflicts, merged, base };
}
