```js
// Commits reachable from compareCommitId but not from baseCommitId, newest first.
export function commitsBetween(db, repoId, baseCommitId, compareCommitId, limit = 250) {
  if (!compareCommitId) return [];
  const baseAncestors = new Set();
  let cursor = baseCommitId;
  while (cursor && !baseAncestors.has(cursor)) {
    baseAncestors.add(cursor);
    const c = commitRow(db, cursor);
    if (!c || c.repo_id !== repoId) break;
    cursor = c.parent_id;
  }
  const out = [];
  cursor = compareCommitId;
  const seen = new Set();
  while (cursor && !seen.has(cursor) && !baseAncestors.has(cursor) && out.length < limit) {
    seen.add(cursor);
    const c = commitRow(db, cursor);
    if (!c || c.repo_id !== repoId) break;
    out.push(decorateCommit(db, c, changedFilesForCommit(db, c).map(f => ({path:f.path,status:f.status,additions:f.additions,deletions:f.deletions}))));
    cursor = c.parent_id;
  }
  return out;
}
```
